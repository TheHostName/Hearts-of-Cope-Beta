on_actions = {
	on_naval_invasion = {
		effect = { 
			THIS = {
				if = {
					limit = {
						NOT = { CONTROLLER = { tag = ROOT } }
						NOT = { CONTROLLER = { has_country_flag = notif_disable_inv_warning } }
					}
					CONTROLLER = {
						set_country_flag = ui_show_invasion_warning
						set_variable = { ui_invaded_state = PREV }
						set_variable = { ui_invaded_country = PREV.owner }
						set_variable = { ui_invader_country = ROOT }
					}
					if = {
						limit = { CONTROLLER = { is_puppet = yes } }
						CONTROLLER = {
							OVERLORD = {
								set_country_flag = ui_show_invasion_warning
								set_variable = { ui_invaded_state = PREV.PREV }
								set_variable = { ui_invaded_country = PREV.PREV.owner }
								set_variable = { ui_invader_country = ROOT }
							}
						}
					} 
					every_country = {
						if = { 
							limit = { 
								PREV.CONTROLLER = { is_in_faction_with = THIS } 
							}
							set_country_flag = ui_show_invasion_warning
							set_variable = { ui_invaded_state = PREV }
							set_variable = { ui_invaded_country = PREV.CONTROLLER }
							set_variable = { ui_invader_country = ROOT }
						}
					}

				}
			}
		}
	}

on_state_control_changed = {
#SOVIET - Clean other state modifiers when Soviets lose control over the state
effect = {
	if = {
		limit = {
			FROM.FROM = { is_core_of = ENG }
			ROOT = { NOT = { tag = ENG } }
		}
		if = { #El-Alamain Battary
			limit = {
				FROM.FROM = { has_dynamic_modifier = { modifier = UK_El_Alamain_battary } }
			}
			FROM.FROM = { remove_dynamic_modifier = { modifier = UK_El_Alamain_battary } }
		}
		if = { #Gibralta Battary
			limit = {
				FROM.FROM = { has_dynamic_modifier = { modifier = UK_Gibralta_battary } }
			}
			FROM.FROM = { remove_dynamic_modifier = { modifier = UK_Gibralta_battary } }
		}

	}
}
